Mysql

ما هو MySQL؟

MySQL هو نظام إدارة قواعد بيانات علائقية (RDBMS) مفتوح المصدر، يُعدّ من أكثر الأنظمة استخدامًا في العالم. يُستخدم في تخزين البيانات وإدارتها باستخدام لغة SQL.

لماذا تتعلم MySQL؟

مجاني ومفتوح المصدر — يمكنك استخدامه بدون تكلفة سريع وموثوق — يُستخدم في شركات عملاقة مثل Facebook وTwitter سهل التعلم — صياغة بسيطة وواضحة دعم واسع — مجتمع كبير ووثائق ممتازة SQL تعني Structured Query Language — لغة الاستعلام الهيكلية. تُستخدم للتواصل مع قواعد البيانات.

أنواع البيانات
إنشاء قاعدة البيانات
      
        -- إنشاء قاعدة بيانات جديدة
        CREATE DATABASE my_store;

        -- عرض جميع قواعد البيانات
        SHOW DATABASES;

        -- اختيار قاعدة البيانات للعمل عليها
        USE my_store;

        -- حذف قاعدة بيانات (تأكد قبل التنفيذ!)
        DROP DATABASE my_store;
الجداول

الجدول هو وحدة تخزين البيانات في MySQL — مثل ورقة Excel لكنها أقوى بكثير.

إنشاء جدول
      
    CREATE TABLE students (
      id         INT AUTO_INCREMENT PRIMARY KEY,
      name       VARCHAR(100) NOT NULL,
      email      VARCHAR(150) UNIQUE,
      age        INT,
      grade      DECIMAL(4,2),
      created_at DATETIME DEFAULT NOW()
    );
أوامر إدارة الجداول
      
        -- عرض جميع الجداول
        SHOW TABLES;

        -- عرض هيكل الجدول
        DESCRIBE students;

        -- إضافة عمود جديد
        ALTER TABLE students ADD phone VARCHAR(20);

        -- حذف عمود
        ALTER TABLE students DROP COLUMN phone;

        -- حذف الجدول كاملًا
        DROP TABLE students;
إدخال البيانات
      
    -- إدخال صف واحد
    INSERT INTO students (name, email, age, grade)
    VALUES ('أحمد محمد', 'ahmed@example.com', 20, 95.5);

    -- إدخال عدة صفوف مرة واحدة
    INSERT INTO students (name, email, age, grade)
    VALUES
      ('سارة علي',    'sara@example.com',  22, 88.0),
      ('محمد يوسف', 'mohd@example.com',  19, 72.5),
      ('نورة خالد',  'noura@example.com', 21, 91.0);
استعلام البيانات
      
    -- استعلام كل البيانات
    SELECT * FROM students;

    -- استعلام أعمدة محددة فقط
    SELECT name, grade FROM students;

    -- ترتيب النتائج تنازليًا
    SELECT name, grade
    FROM students
    ORDER BY grade DESC;

    -- تحديد عدد النتائج
    SELECT * FROM students LIMIT 5;

    -- إزالة التكرارات
    SELECT DISTINCT age FROM students;
التحديث والحذف

دائمًا استخدم WHERE مع UPDATE وDELETE، وإلا ستؤثر العملية على كل الصفوف!

      
    -- تحديث بيانات طالب
    UPDATE students
    SET grade = 98.0, age = 21
    WHERE id = 1;

    -- حذف سجل واحد
    DELETE FROM students
    WHERE id = 3;

    -- حذف بشرط متعددة
    DELETE FROM students
    WHERE grade < 50;

    -- حذف كل البيانات (مع الإبقاء على الجدول)
    TRUNCATE TABLE students;
الشروط والفلترة
      
    -- شرط مقارنة بسيط
    SELECT * FROM students WHERE age > 20;

    -- AND / OR
    SELECT * FROM students
    WHERE age >= 18 AND grade >= 80;

    -- BETWEEN — بين قيمتين
    SELECT * FROM students
    WHERE grade BETWEEN 70 AND 90;

    -- LIKE — البحث بنمط
    SELECT * FROM students
    WHERE name LIKE 'أحمد%';     -- يبدأ بـ أحمد

    SELECT * FROM students
    WHERE email LIKE '%@gmail.com'; -- ينتهي بـ gmail

    -- IN — ضمن قائمة
    SELECT * FROM students
    WHERE age IN (18, 19, 20);

    -- IS NULL — القيم الفارغة
    SELECT * FROM students
    WHERE email IS NULL;
الدوال المدمجة
      
    -- COUNT — عدد الصفوف
    SELECT COUNT(*) AS total_students FROM students;

    -- SUM / AVG / MAX / MIN
    SELECT
      AVG(grade)  AS average_grade,
      MAX(grade)  AS highest_grade,
      MIN(grade)  AS lowest_grade,
      SUM(grade)  AS total_grades
    FROM students;

    -- GROUP BY — تجميع
    SELECT age, COUNT(*) AS count
    FROM students
    GROUP BY age;

    -- HAVING — فلترة بعد التجميع
    SELECT age, AVG(grade) AS avg_grade
    FROM students
    GROUP BY age
    HAVING AVG(grade) > 80;
دوال النصوص
      
    SELECT
      UPPER(name)          AS upper_name,  -- أحرف كبيرة
      LOWER(name)          AS lower_name,  -- أحرف صغيرة
      LENGTH(name)         AS name_len,   -- طول النص
      CONCAT(name, ' 👨‍🎓') AS label       -- دمج النصوص
    FROM students;
الربط بين الجداول

JOIN — المفهوم الأهم في SQL

JOIN يُمكّنك من دمج بيانات من جدولين أو أكثر بناءً على عمود مشترك.

أنواع JOIN الأساسية: INNER JOIN (متطابق)، LEFT JOIN (كل اليسار)، RIGHT JOIN (كل اليمين)

      
    -- أولًا نُنشئ جدول الدورات
    CREATE TABLE courses (
      id         INT AUTO_INCREMENT PRIMARY KEY,
      student_id INT,
      course     VARCHAR(100),
      score      INT
    );

    -- INNER JOIN — السجلات المطابقة في الجدولين
    SELECT
      students.name,
      courses.course,
      courses.score
    FROM students
    INNER JOIN courses
      ON students.id = courses.student_id;

    -- LEFT JOIN — كل الطلاب حتى بدون دورات
    SELECT
      students.name,
      COALESCE(courses.course, 'لا يوجد') AS course
    FROM students
    LEFT JOIN courses
      ON students.id = courses.student_id;
